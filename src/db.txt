-- STEP 1: Create the database
CREATE DATABASE GreenRecycleDB;

-- STEP 2: Select the database
USE GreenRecycleDB;


-- STEP 3: Create range-partitioned table
-- Collection_Date is included in the PRIMARY KEY
-- because it is the partitioning column.

CREATE TABLE EWaste_Management (
    Waste_ID INT,
    Device_Type VARCHAR(50),
    Manufacturer VARCHAR(50),
    Collection_Date DATE,
    Collection_Center_ID INT,
    Weight DECIMAL(8,2),
    Status VARCHAR(20),

    PRIMARY KEY (Waste_ID, Collection_Date)
)
PARTITION BY RANGE (YEAR(Collection_Date))
(
    PARTITION p2022 VALUES LESS THAN (2023),
    PARTITION p2023 VALUES LESS THAN (2024),
    PARTITION p2024 VALUES LESS THAN (2025),
    PARTITION p2025 VALUES LESS THAN (2026),
    PARTITION pFuture VALUES LESS THAN MAXVALUE
);

-- STEP 6: Query records from a specific year
-- This retrieves records collected in 2024.

SELECT *
FROM EWaste_Management
WHERE Collection_Date >= '2024-01-01'
  AND Collection_Date < '2025-01-01';


-- STEP 7: Check partition information

SELECT
    PARTITION_NAME,
    TABLE_ROWS
FROM INFORMATION_SCHEMA.PARTITIONS
WHERE TABLE_SCHEMA = 'GreenRecycleDB'
  AND TABLE_NAME = 'EWaste_Management';


-- STEP 8: Check which partition contains the records
-- This is optional but useful for demonstration.

SELECT
    TABLE_NAME,
    PARTITION_NAME,
    TABLE_ROWS
FROM INFORMATION_SCHEMA.PARTITIONS
WHERE TABLE_SCHEMA = 'GreenRecycleDB'
  AND TABLE_NAME = 'EWaste_Management'
  AND PARTITION_NAME IS NOT NULL;


-- STEP 9: Query another specific year

SELECT *
FROM EWaste_Management
WHERE Collection_Date >= '2023-01-01'
  AND Collection_Date < '2024-01-01';


-- STEP 10: Query records from Future partition
-- 2026 record will be stored in pFuture.

SELECT *
FROM EWaste_Management
WHERE Collection_Date >= '2026-01-01';




hash partitioning

CREATE DATABASE GreenRecycleDB;

USE GreenRecycleDB;

CREATE TABLE EWaste_Management (
    Waste_ID INT,
    Device_Type VARCHAR(50),
    Manufacturer VARCHAR(50),
    Collection_Date DATE,
    Collection_Center_ID INT,
    Weight DECIMAL(8,2),
    Status VARCHAR(20),
    PRIMARY KEY (Waste_ID, Collection_Center_ID)
)
PARTITION BY HASH(Collection_Center_ID)
PARTITIONS 4;
